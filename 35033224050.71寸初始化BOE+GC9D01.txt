void Lcd_enter_sleep(void)
{
	LCD_CtrlWrite(0x28);
	DelayMs(120);
	LCD_CtrlWrite(0x10);
	DelayMs(50);
}
void Lcd_exit_sleep(void)
{
	LCD_CtrlWrite(0x11);
	DelayMs(120) ;
	LCD_CtrlWrite(0x29);
}
